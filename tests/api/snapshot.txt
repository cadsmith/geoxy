== geoxy/angle.hpp
#include "geoxy/api.hpp"
#include "geoxy/vector.hpp"
#include <compare>
#include <functional>
#include <numbers>
namespace Geoxy
{
  inline constexpr double Pi = std::numbers::pi;
  class Angle
  {
  public:
    constexpr Angle() = default;
    [[nodiscard]] constexpr auto InRadians() const -> double
    {
      return m_InDegrees ? Converted(m_Value, true) : m_Value;
    }
    [[nodiscard]] constexpr auto InDegrees() const -> double
    {
      return m_InDegrees ? m_Value : Converted(m_Value, false);
    }
    [[nodiscard]] constexpr auto HeldInDegrees() const -> bool { return m_InDegrees; }
    [[nodiscard]] friend constexpr auto operator==(const Angle &lhs, const Angle &rhs) -> bool
    {
      if (lhs.m_InDegrees == rhs.m_InDegrees)
      {
        return std::equal_to<double>{}(lhs.m_Value, rhs.m_Value);
      }
      return std::equal_to<double>{}(lhs.InDegrees(), rhs.InDegrees());
    }
    [[nodiscard]] friend constexpr auto operator<=>(const Angle &lhs, const Angle &rhs)
        -> std::partial_ordering
    {
      if (lhs.m_InDegrees == rhs.m_InDegrees)
      {
        return lhs.m_Value <=> rhs.m_Value;
      }
      return lhs.InDegrees() <=> rhs.InDegrees();
    }
    friend constexpr auto Radians(double radians) -> Angle;
    friend constexpr auto Degrees(double degrees) -> Angle;
  private:
    constexpr Angle(double value, bool inDegrees) : m_Value(value), m_InDegrees(inDegrees) {}
    [[nodiscard]] GEOXY_API static auto Converted(double value, bool toRadians) -> double;
    double m_Value     = 0.0;   
    bool   m_InDegrees = false; 
  };
  [[nodiscard]] constexpr auto Radians(double radians) -> Angle
  {
    return {radians, false};
  }
  [[nodiscard]] constexpr auto Degrees(double degrees) -> Angle
  {
    return {degrees, true};
  }
  inline constexpr Angle Turn = Degrees(360.0);
  [[nodiscard]] constexpr auto operator+(Angle lhs, Angle rhs) -> Angle
  {
    if (lhs.HeldInDegrees() || rhs.HeldInDegrees())
    {
      return Degrees(lhs.InDegrees() + rhs.InDegrees());
    }
    return Radians(lhs.InRadians() + rhs.InRadians());
  }
  [[nodiscard]] constexpr auto operator-(Angle lhs, Angle rhs) -> Angle
  {
    if (lhs.HeldInDegrees() || rhs.HeldInDegrees())
    {
      return Degrees(lhs.InDegrees() - rhs.InDegrees());
    }
    return Radians(lhs.InRadians() - rhs.InRadians());
  }
  [[nodiscard]] constexpr auto operator-(Angle angle) -> Angle
  {
    return angle.HeldInDegrees() ? Degrees(-angle.InDegrees()) : Radians(-angle.InRadians());
  }
  [[nodiscard]] constexpr auto operator*(Angle lhs, double rhs) -> Angle
  {
    return lhs.HeldInDegrees() ? Degrees(lhs.InDegrees() * rhs) : Radians(lhs.InRadians() * rhs);
  }
  [[nodiscard]] constexpr auto operator/(Angle lhs, double rhs) -> Angle
  {
    return lhs.HeldInDegrees() ? Degrees(lhs.InDegrees() / rhs) : Radians(lhs.InRadians() / rhs);
  }
  [[nodiscard]] GEOXY_API auto Wrapped(Angle angle) -> Angle;
  [[nodiscard]] GEOXY_API auto Within(Angle direction, Angle start, Angle sweep) -> bool;
  [[nodiscard]] GEOXY_API auto Direction(Angle angle) -> Vector;
  [[nodiscard]] GEOXY_API auto Direction(Angle start, Angle sweep, double along) -> Vector;
  [[nodiscard]] GEOXY_API auto Heading(Vector vector) -> Angle;
  [[nodiscard]] GEOXY_API auto Between(Vector start, Vector end) -> Angle;
  [[nodiscard]] GEOXY_API auto Order(Vector lhs, Vector rhs) -> std::strong_ordering;
}

== geoxy/box.hpp
#include "geoxy/api.hpp"
#include "geoxy/point.hpp"
#include "geoxy/vector.hpp"
#include <algorithm>
#include <limits>
namespace Geoxy
{
  struct Box
  {
    Point Min = {.X = std::numeric_limits<double>::infinity(),
                 .Y = std::numeric_limits<double>::infinity()};
    Point Max = {.X = -std::numeric_limits<double>::infinity(),
                 .Y = -std::numeric_limits<double>::infinity()};
    [[nodiscard]] friend constexpr auto operator==(const Box &lhs, const Box &rhs) -> bool
    {
      return lhs.Min == rhs.Min && lhs.Max == rhs.Max;
    }
  };
  [[nodiscard]] constexpr auto IsEmpty(const Box &box) -> bool
  {
    return !(box.Min.X <= box.Max.X) || !(box.Min.Y <= box.Max.Y);
  }
  [[nodiscard]] constexpr auto Grown(const Box &box, Point point) -> Box
  {
    return {.Min = {.X = std::min(box.Min.X, point.X), .Y = std::min(box.Min.Y, point.Y)},
            .Max = {.X = std::max(box.Max.X, point.X), .Y = std::max(box.Max.Y, point.Y)}};
  }
  [[nodiscard]] constexpr auto Joined(const Box &first, const Box &second) -> Box
  {
    const Point least = {.X = std::min(first.Min.X, second.Min.X),
                         .Y = std::min(first.Min.Y, second.Min.Y)};
    const Point most  = {.X = std::max(first.Max.X, second.Max.X),
                         .Y = std::max(first.Max.Y, second.Max.Y)};
    return {.Min = least, .Max = most};
  }
  [[nodiscard]] constexpr auto Contains(const Box &box, Point point) -> bool
  {
    return box.Min.X <= point.X
        && point.X   <= box.Max.X
        && box.Min.Y <= point.Y
        && point.Y   <= box.Max.Y;
  }
  [[nodiscard]] constexpr auto Overlaps(const Box &first, const Box &second) -> bool
  {
    return first.Min.X  <= second.Max.X
        && second.Min.X <= first.Max.X
        && first.Min.Y  <= second.Max.Y
        && second.Min.Y <= first.Max.Y;
  }
  [[nodiscard]] constexpr auto Size(const Box &box) -> Vector
  {
    return box.Max - box.Min;
  }
  [[nodiscard]] GEOXY_API auto Center(const Box &box) -> Point;
  [[nodiscard]] GEOXY_API auto Padded(const Box &box, double margin) -> Box;
}

== geoxy/point.hpp
#include "geoxy/vector.hpp"
#include <functional>
#include <numeric>
namespace Geoxy
{
  struct Point
  {
    double X = 0.0; 
    double Y = 0.0; 
    [[nodiscard]] friend constexpr auto operator==(const Point &lhs, const Point &rhs) -> bool
    {
      return std::equal_to<double>{}(lhs.X, rhs.X) && std::equal_to<double>{}(lhs.Y, rhs.Y);
    }
  };
  [[nodiscard]] constexpr auto operator-(Point lhs, Point rhs) -> Vector
  {
    return {.X = lhs.X - rhs.X, .Y = lhs.Y - rhs.Y};
  }
  [[nodiscard]] constexpr auto operator+(Point lhs, Vector rhs) -> Point
  {
    return {.X = lhs.X + rhs.X, .Y = lhs.Y + rhs.Y};
  }
  [[nodiscard]] constexpr auto operator-(Point lhs, Vector rhs) -> Point
  {
    return {.X = lhs.X - rhs.X, .Y = lhs.Y - rhs.Y};
  }
  [[nodiscard]] constexpr auto Midpoint(Point first, Point second) -> Point
  {
    return {.X = std::midpoint(first.X, second.X), .Y = std::midpoint(first.Y, second.Y)};
  }
  [[nodiscard]] constexpr auto Lerp(Point start, Point end, double fraction) -> Point
  {
    const Vector across = end - start;
    if (fraction <= 0.5)
    {
      return start + (across * fraction);
    }
    return end - (across * (1.0 - fraction));
  }
  [[nodiscard]] inline auto Distance(Point first, Point second) -> double
  {
    return Length(second - first);
  }
}

== geoxy/vector.hpp
#include "geoxy/api.hpp"
#include <algorithm>
#include <cmath>
#include <functional>
#include <optional>
namespace Geoxy
{
  struct Vector
  {
    double X = 0.0; 
    double Y = 0.0; 
    [[nodiscard]] friend constexpr auto operator==(const Vector &lhs, const Vector &rhs) -> bool
    {
      return std::equal_to<double>{}(lhs.X, rhs.X) && std::equal_to<double>{}(lhs.Y, rhs.Y);
    }
  };
  [[nodiscard]] constexpr auto operator+(Vector lhs, Vector rhs) -> Vector
  {
    return {.X = lhs.X + rhs.X, .Y = lhs.Y + rhs.Y};
  }
  [[nodiscard]] constexpr auto operator-(Vector lhs, Vector rhs) -> Vector
  {
    return {.X = lhs.X - rhs.X, .Y = lhs.Y - rhs.Y};
  }
  [[nodiscard]] constexpr auto operator-(Vector vector) -> Vector
  {
    return {.X = -vector.X, .Y = -vector.Y};
  }
  [[nodiscard]] constexpr auto operator*(Vector lhs, double rhs) -> Vector
  {
    return {.X = lhs.X * rhs, .Y = lhs.Y * rhs};
  }
  [[nodiscard]] constexpr auto operator*(double lhs, Vector rhs) -> Vector
  {
    return rhs * lhs;
  }
  [[nodiscard]] constexpr auto operator/(Vector lhs, double rhs) -> Vector
  {
    return {.X = lhs.X / rhs, .Y = lhs.Y / rhs};
  }
  [[nodiscard]] constexpr auto Perpendicular(Vector vector) -> Vector
  {
    return {.X = -vector.Y, .Y = vector.X};
  }
  [[nodiscard]] constexpr auto Largest(Vector vector) -> double
  {
    const double x = vector.X < 0.0 ? -vector.X : vector.X;
    const double y = vector.Y < 0.0 ? -vector.Y : vector.Y;
    return std::max(x, y);
  }
  [[nodiscard]] constexpr auto Dot(Vector first, Vector second) -> double
  {
    return (first.X * second.X) + (first.Y * second.Y);
  }
  [[nodiscard]] constexpr auto Cross(Vector first, Vector second) -> double
  {
    return (first.X * second.Y) - (first.Y * second.X);
  }
  [[nodiscard]] inline auto Length(Vector vector) -> double
  {
    constexpr double Low     = 0x1.0p-500;
    constexpr double High    = 0x1.0p500;
    const double     largest = Largest(vector);
    if (largest >= Low && largest <= High)
    {
      return std::sqrt(Dot(vector, vector));
    }
    if (!(largest > 0.0) || !std::isfinite(largest))
    {
      return largest;
    }
    const Vector scaled = vector / largest;
    return largest * std::sqrt(Dot(scaled, scaled));
  }
  [[nodiscard]] GEOXY_API auto Unit(Vector vector) -> std::optional<Vector>;
}
